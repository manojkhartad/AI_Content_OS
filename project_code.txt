import os

# Agar script project ke root me hai to "." use karo
ROOT_DIR = "."

OUTPUT_FILE = "project_code.txt"

# Jin folders ko ignore karna hai
IGNORE_DIRS = {
    "__pycache__",
    ".git",
    ".venv",
    "venv",
    "env",
    "node_modules",
    "build",
    "dist",
    ".idea",
    ".vscode"
}

with open(OUTPUT_FILE, "w", encoding="utf-8") as outfile:

    for foldername, subfolders, filenames in os.walk(ROOT_DIR):

        # Ignore folders
        subfolders[:] = [d for d in subfolders if d not in IGNORE_DIRS]

        for filename in sorted(filenames):

            if filename.endswith(".py"):

                file_path = os.path.join(foldername, filename)
                relative_path = os.path.relpath(file_path, ROOT_DIR)

                outfile.write("=" * 58 + "\n")
                outfile.write(f"FILE: {relative_path}\n")
                outfile.write("=" * 58 + "\n\n")

                try:
                    with open(file_path, "r", encoding="utf-8") as f:
                        outfile.write(f.read())
                except Exception as e:
                    outfile.write(f"Error reading file: {e}")

                outfile.write("\n\n")

print(f"Done! Output saved to '{OUTPUT_FILE}'")

                    voice="en-US-AriaNeural",

import re
from pathlib import Path

from agents.base_agent import BaseAgent
from brain.video_client import VideoClient
from config.paths import VIDEOS_DIR
from exporters.video_exporter import VideoExporter
from models.generated_video import GeneratedVideo
from config.settings import (
    VIDEO_WIDTH,
    VIDEO_HEIGHT,
    VIDEO_FPS,
)


# ==========================================================
# Video Generation Agent
# ==========================================================

class VideoGenerationAgent(BaseAgent):

    def __init__(self):

        super().__init__("Video Generation Agent")

    def generate_video(

        self,

        draft,

        generated_images,

        generated_audio

    ):

        print("Video Generation Started")

        scene_videos = []

        # --------------------------------------------------
        # Create Scene Videos
        # --------------------------------------------------

        for image, audio in zip(
            generated_images,
            generated_audio
        ):

            print(
                f"\nGenerating Video : Scene {image.scene_number}"
            )

            output_path = (
                VIDEOS_DIR /
                f"scene_{image.scene_number:03d}.mp4"
            )

            try:

                VideoClient.create_scene_video(

                    image_path=Path(
                        image.image_path
                    ),

                    audio_path=Path(
                        audio.audio_path
                    ),

                    output_path=output_path

                )

                scene_videos.append(
                    output_path
                )

                print(
                    f"✓ Scene {image.scene_number} Video Created"
                )

            except Exception as e:

                print(
                    f"\n❌ Failed : Scene {image.scene_number}"
                )

                print(e)

                continue

        # --------------------------------------------------
        # Stop if no videos were created
        # --------------------------------------------------

        if not scene_videos:

            raise RuntimeError(
                "No scene videos were created."
            )

        # --------------------------------------------------
        # Create Safe File Name
        # --------------------------------------------------

        safe_title = re.sub(
            r"[^A-Za-z0-9_-]+",
            "_",
            draft.title
        ).strip("_")

        # --------------------------------------------------
        # Merge Videos
        # --------------------------------------------------

        print("\nCreating Final Video...")

        final_video = (
            VIDEOS_DIR /
            f"{safe_title}.mp4"
        )

        VideoClient.merge_scene_videos(

            video_paths=scene_videos,

            output_path=final_video

        )

        # --------------------------------------------------
        # Calculate Total Duration
        # --------------------------------------------------

        total_duration = sum(
            audio.duration
            for audio in generated_audio
        )

        # --------------------------------------------------
        # Video Metadata
        # --------------------------------------------------

        video = GeneratedVideo(

            title=draft.title,

            video_path=str(
                final_video
            ),

            width=VIDEO_WIDTH,

            height=VIDEO_HEIGHT,

            fps=VIDEO_FPS,

            duration=round(
                total_duration,
                2
            ),

            codec="H.264",

            format="MP4",

            provider="FFmpeg",

            total_scenes=len(scene_videos),

            status="completed"

        )

        VideoExporter.export(
            video
        )

        print("\n===================================")
        print("Video Generation Completed")
        print(f"Saved : {final_video}")
        print("===================================")

        return video

from pathlib import Path
from urllib.parse import quote

import requests


# ==========================================================
# Image Client
# ==========================================================

class ImageClient:

    BASE_URL = "https://image.pollinations.ai/prompt/"


    @staticmethod
    def generate_image(
        prompt: str,
        output_path: Path,
        width: int = 1024,
        height: int = 576,
    ) -> Path:

        output_path.parent.mkdir(
            parents=True,
            exist_ok=True
        )

        encoded_prompt = quote(prompt)

        image_url = (
            f"{ImageClient.BASE_URL}"
            f"{encoded_prompt}"
            f"?width={width}"
            f"&height={height}"
            f"&model=flux"
        )

        print("\nGenerating Image...")
        print(image_url)

        response = requests.get(
            image_url,
            timeout=180
        )

        response.raise_for_status()

        with open(output_path, "wb") as file:
            file.write(response.content)

        print(f"Image Saved : {output_path}")

        return output_path

from google import genai
from google.genai.errors import ClientError, ServerError

from config.settings import GEMINI_API_KEY


class LLMClient:

    def __init__(self):
        self.client = genai.Client(api_key=GEMINI_API_KEY)

        self.models = [
            "models/gemma-4-31b-it",
            "models/gemini-3.5-flash",
            "models/gemini-3.1-flash-lite",
        ]

    def generate(self, prompt):

        last_error = None

        for model in self.models:

            try:

                print(f"\nTrying Model : {model}")

                response = self.client.models.generate_content(
                    model=model,
                    contents=prompt,
                )

                print(f"Success : {model}")

                return response.text

            except (ClientError, ServerError) as e:

                print(f"Failed : {model}")
                print(e)

                last_error = e

        raise last_error

import asyncio
from pathlib import Path

import edge_tts

from config.settings import TTS_PROVIDER


# ==========================================================
# TTS Client
# ==========================================================

class TTSClient:

    DEFAULT_VOICE = "en-US-AvaNeural"

    @staticmethod
    async def _generate(
        text: str,
        output_path: Path,
        voice: str
    ):

        communicate = edge_tts.Communicate(
            text=text,
            voice=voice
        )

        await communicate.save(
            str(output_path)
        )

    @staticmethod
    def generate_audio(
        text: str,
        output_path: Path,
        voice: str = None
    ):

        if TTS_PROVIDER != "edge":
            raise ValueError(
                f"Unsupported TTS Provider : {TTS_PROVIDER}"
            )

        if voice is None:
            voice = TTSClient.DEFAULT_VOICE

        output_path.parent.mkdir(
            parents=True,
            exist_ok=True
        )

        print("\nGenerating Audio...")

        asyncio.run(
            TTSClient._generate(
                text=text,
                output_path=output_path,
                voice=voice
            )
        )

        print(f"Audio Saved : {output_path}")

        return output_path

import subprocess
from pathlib import Path

from config.settings import (
    FFMPEG_PATH,
    VIDEO_FPS,
    VIDEO_WIDTH,
    VIDEO_HEIGHT,
)


# ==========================================================
# Video Client (FFmpeg)
# ==========================================================

class VideoClient:

    # ------------------------------------------------------
    # Image + Audio -> Scene Video
    # ------------------------------------------------------

    @staticmethod
    def create_scene_video(

        image_path: Path,
        audio_path: Path,
        output_path: Path

    ):

        print("\nGenerating Scene Video...")

        command = [

            FFMPEG_PATH,

            "-y",

            "-loop", "1",

            "-i", str(image_path),

            "-i", str(audio_path),

            "-vf",

            (
                f"scale={VIDEO_WIDTH}:{VIDEO_HEIGHT},"
                f"zoompan="
                f"z='min(zoom+0.0008,1.15)':"
                f"d=10000:"
                f"x='iw/2-(iw/zoom/2)':"
                f"y='ih/2-(ih/zoom/2)',"
                f"fps={VIDEO_FPS}"
            ),

            "-c:v", "libx264",

            "-preset", "medium",

            "-tune", "stillimage",

            "-pix_fmt", "yuv420p",

            "-c:a", "aac",

            "-b:a", "192k",

            "-shortest",

            "-movflags", "+faststart",

            str(output_path)

        ]

        subprocess.run(
            command,
            check=True
        )

        print(
            f"Scene Video Saved : {output_path}"
        )

    # ------------------------------------------------------
    # Merge Scene Videos
    # ------------------------------------------------------

    @staticmethod
    def merge_scene_videos(

        video_paths: list[Path],

        output_path: Path

    ):

        print("\nMerging Scene Videos...")

        list_file = output_path.parent / "videos.txt"

        with open(

            list_file,

            "w",

            encoding="utf-8"

        ) as file:

            for video in video_paths:

                file.write(
                    f"file '{video.resolve()}'\n"
                )

        command = [

            FFMPEG_PATH,

            "-y",

            "-f", "concat",

            "-safe", "0",

            "-i", str(list_file),

            "-c", "copy",

            "-movflags", "+faststart",

            str(output_path)

        ]

        subprocess.run(

            command,

            check=True

        )

        if list_file.exists():

            list_file.unlink()

        print(
            f"Final Video Saved : {output_path}"
        )

class AudioExporter:

    @staticmethod
    def export(audio: GeneratedAudio):

        output_file = (
            AUDIO_DIR /
            f"scene_{audio.scene_number:03d}.json"
        )

        data = {

            "scene_id": audio.scene_id,

            "scene_number": audio.scene_number,

            "title": audio.title,

            "narration": audio.narration,

            "audio_path": audio.audio_path,

            "duration": audio.duration,

            "voice": audio.voice,

            "provider": audio.provider,

            "model": audio.model,

            "sample_rate": audio.sample_rate,

            "format": audio.format

        }

        with open(
            output_file,
            "w",
            encoding="utf-8"
        ) as f:

            json.dump(
                data,
                f,
                indent=4,
                ensure_ascii=False
            )

        print("\nAudio Exported Successfully")

        print(f"Scene : {audio.scene_number}")

        print(f"Saved : {output_file}")

        )

        data = {
            "title": narration_list.title,

                "scene_id": narration.scene_id,

                "scene_number": narration.scene_number,

                "title": narration.title,

                "narration": narration.narration,

                "word_count": narration.word_count,

                "estimated_duration": narration.estimated_duration,

                "audio_file": narration.audio_file,

                "subtitle_file": narration.subtitle_file

            })

        with open(
            output_file,
            "w",
            encoding="utf-8"
        ) as file:

            json.dump(
                data,
                file,
                indent=4,
                ensure_ascii=False
            )

        print("\nNarrations exported successfully.")
        print(output_file)

        return output_file

import json


def parse_json(response: str) -> dict:
    """
    Parse JSON returned by the LLM.

    Supports:
    - Plain JSON
    - ```json ... ```
    - ``` ... ```

    Returns:
        dict

    Raises:
        ValueError if JSON is invalid.
    """

    if not response:
        raise ValueError("LLM returned an empty response.")

    cleaned = response.strip()

    # Remove ```json
    if cleaned.startswith("```json"):
        cleaned = cleaned.replace("```json", "", 1)

    # Remove ```
    if cleaned.startswith("```"):
        cleaned = cleaned.replace("```", "", 1)

    if cleaned.endswith("```"):
        cleaned = cleaned[:-3]

    cleaned = cleaned.strip()

    try:
        return json.loads(cleaned)

    except json.JSONDecodeError as e:
        raise ValueError(
            f"Invalid JSON returned by LLM.\n\nResponse:\n{response}"
        ) from e

class StoryWorkflow:

    def run(self, topic):

        print("Story Workflow Started")

        # --------------------------------------------------
        # Research
        # --------------------------------------------------

        research_agent = ResearchAgent()
        research_agent.start()

        research_report = research_agent.research(topic)

        # --------------------------------------------------
        # Character Profiles
        # --------------------------------------------------

        character_agent = CharacterAgent()
        character_agent.start()

        characters = character_agent.generate_characters(
            research_report
        )

        # --------------------------------------------------
        # Character Bible
        # --------------------------------------------------

        character_bible_agent = CharacterBibleAgent()
        character_bible_agent.start()

        character_bibles = (
            character_bible_agent.generate_character_bibles(
                characters
            )
        )

        # --------------------------------------------------

        # --------------------------------------------------

        CharacterBibleExporter.export(
            character_bibles
        )

        # Story Writer
        # --------------------------------------------------

        writer_agent = WriterAgent()
        writer_agent.start()

        draft = writer_agent.write(
            research_report,
            character_bibles
        )

        # --------------------------------------------------
        # Story Reviewer
        # --------------------------------------------------

        reviewer_agent.start()

        review = reviewer_agent.review(
            draft
        )

        StoryFormatter.print_review(
            review
        )

        # --------------------------------------------------
        # Export Story
        # --------------------------------------------------

        StoryExporter.export(
            draft
        )

        # --------------------------------------------------
        # Scene Generation
        # --------------------------------------------------

        scene_agent = SceneAgent()
        scene_agent.start()

        scene_list = scene_agent.generate_scenes(
            draft,
            character_bibles
        )

        # --------------------------------------------------
        # Export Scenes
        # --------------------------------------------------

        SceneExporter.export(
            scene_list
        )

        # --------------------------------------------------
        # Narration Generation
        # --------------------------------------------------

        narration_agent = NarrationAgent()
        narration_agent.start()

        narration_list = (
            narration_agent.generate_narrations(
                scene_list
            )
        )

        # --------------------------------------------------
        # Export Narrations
        # --------------------------------------------------

        NarrationExporter.export(
            narration_list
        )

        # --------------------------------------------------
        # Image Prompt Generation
        # --------------------------------------------------

        image_prompt_agent = ImagePromptAgent()
        image_prompt_agent.start()

        image_prompts = (
            image_prompt_agent.generate_image_prompts(
                scene_list.scenes,
                character_bibles
            )
        )

        # --------------------------------------------------
        # Export Image Prompts
        # --------------------------------------------------

        ImagePromptExporter.export(
            image_prompts
        )

        # --------------------------------------------------
        # Image Generation
        # --------------------------------------------------

        image_generation_agent = ImageGenerationAgent()
        image_generation_agent.start()

        generated_images = (
            image_generation_agent.generate_images(
                image_prompts
            )
        )

        # --------------------------------------------------
        # Audio Generation (TTS)
        # --------------------------------------------------

        tts_agent = TTSAgent()
        tts_agent.start()

        generated_audio = (
            tts_agent.generate_audio(
